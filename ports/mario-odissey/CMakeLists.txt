cmake_minimum_required(VERSION 3.16)
project(mgd_port_odyssey VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

file(GLOB_RECURSE PORT_SOURCES "src/mgd/*.cpp")

add_library(mgd_port STATIC ${PORT_SOURCES})
target_include_directories(mgd_port PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)

add_executable(mgd_port_check tools/check.cpp)
target_link_libraries(mgd_port_check PRIVATE mgd_port)

add_executable(port_tests_nso tests/test_nso_header.cpp)
target_link_libraries(port_tests_nso PRIVATE mgd_port)
add_executable(port_tests_nca tests/test_nca_header.cpp)
target_link_libraries(port_tests_nca PRIVATE mgd_port)
add_executable(port_tests_patch tests/test_patch_manifest.cpp)
target_link_libraries(port_tests_patch PRIVATE mgd_port)
add_executable(port_tests_romfs tests/test_romfs_analyzers.cpp)
target_link_libraries(port_tests_romfs PRIVATE mgd_port)
add_executable(port_tests_services tests/test_service_registry.cpp)
target_link_libraries(port_tests_services PRIVATE mgd_port)
add_executable(port_tests_fs tests/test_virtual_fs.cpp)
target_link_libraries(port_tests_fs PRIVATE mgd_port)
add_executable(port_tests_clock tests/test_system_clock.cpp)
target_link_libraries(port_tests_clock PRIVATE mgd_port)
add_executable(port_tests_input tests/test_input.cpp)
target_link_libraries(port_tests_input PRIVATE mgd_port)
add_executable(port_tests_npdm tests/test_npdm_header.cpp)
target_link_libraries(port_tests_npdm PRIVATE mgd_port)
add_executable(nca_list tools/nca_list.cpp)
target_link_libraries(nca_list PRIVATE mgd_port)
